local function makeLabel(parent, text, size, pos, font, color)

-- Drag
do
    local dragging = false
    local dragStart, startPos
    header.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = root.Position
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            root.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end

local function toggle(parent, text1, text2, y, default)
    local label = makeLabel(parent, text1, 15, UDim2.new(0,14,0,y), Enum.Font.GothamSemibold)
    label.Size = UDim2.new(1,-75,0,22)
    local sub = makeLabel(parent, text2 or "", 12, UDim2.new(0,14,0,y+20), Enum.Font.Gotham, MUTED)
    sub.Size = UDim2.new(1,-75,0,18)

    local b = Instance.new("TextButton")
    b.Text = ""
    b.AutoButtonColor = false
    b.BackgroundColor3 = default == false and Color3.fromRGB(50,52,60) or RED
    b.Position = UDim2.new(1,-48,0,y+5)
    b.Size = UDim2.new(0,38,0,24)
    b.Parent = parent
    local bc = Instance.new("UICorner")
    bc.CornerRadius = UDim.new(1,0)
    bc.Parent = b

    local knob = Instance.new("Frame")
    knob.Size = UDim2.new(0,18,0,18)
    knob.Position = default == false and UDim2.new(0,3,0,3) or UDim2.new(1,-21,0,3)
    knob.BackgroundColor3 = Color3.fromRGB(245,247,255)
    knob.Parent = b
    local kc = Instance.new("UICorner")
    kc.CornerRadius = UDim.new(1,0)
    kc.Parent = knob

    local state = default ~= false
    local function render()
        b.BackgroundColor3 = state and RED or Color3.fromRGB(50,52,60)
        knob:TweenPosition(state and UDim2.new(1,-21,0,3) or UDim2.new(0,3,0,3), "Out", "Quad", 0.12, true)
    end
    b.MouseButton1Click:Connect(function()
        state = not state
        render()
        -- Hook: connect state changes to your own server-authorized feature here.
    end)
    return function() return state end
end

local function dropdown(parent, labelText, y, value, options)

    b.Position = UDim2.new(0,14,0,y+25)
    b.Size = UDim2.new(1,-28,0,34)
    b.Parent = parent
    local c = Instance.new("UICorner"); c.CornerRadius = UDim.new(0,7); c.Parent = b
    stroke(b, Color3.fromRGB(62,65,79), 1, 0)
    local index = table.find(options,value) or 1
    b.MouseButton1Click:Connect(function()
        index = index % #options + 1
        b.Text = options[index].."   ˅"
    end)
    return b
end

local function genericPage(name, sections)
    local page = pages[name]
    local n = #sections
    for i,sec in ipairs(sections) do
        local cols = n > 4 and 3 or 2
        local row = math.floor((i-1)/cols)
        local col = (i-1)%cols
        local w = 0.31
        local x = 0.02 + col*0.325
        local y = 0.02 + row*0.32
        local p = panel(page,sec[1],x,y,w,0.29,sec[2])
        local yy=53
        for _,item in ipairs(sec[3]) do
            if item.kind=="toggle" then
                toggle(p,item.a,item.b,yy,true); yy=yy+47
            elseif item.kind=="drop" then
                dropdown(p,item.a,yy,item.b,item.c); yy=yy+70
            end
        end
    end
end
